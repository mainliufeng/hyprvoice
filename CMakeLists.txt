cmake_minimum_required(VERSION 3.22)
project(hyprvoice VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4 gtk4-layer-shell-0)
pkg_check_modules(PW REQUIRED IMPORTED_TARGET libpipewire-0.3)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_library(SHERPA sherpa-onnx-c-api REQUIRED)
add_library(hyprvoice_json INTERFACE)
target_include_directories(hyprvoice_json INTERFACE third_party)
find_package(Threads REQUIRED)
add_library(hyprvoice_core src/config.cpp src/process.cpp src/asr.cpp src/vad_trimmer.cpp src/audio.cpp src/desktop.cpp src/rewrite.cpp)
target_include_directories(hyprvoice_core PUBLIC src)
target_link_libraries(hyprvoice_core PUBLIC PkgConfig::GTK PkgConfig::PW PkgConfig::CURL ${SHERPA} hyprvoice_json Threads::Threads)
target_compile_options(hyprvoice_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(hyprvoice src/main.cpp src/app.cpp)
target_link_libraries(hyprvoice PRIVATE hyprvoice_core)
target_compile_options(hyprvoice PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/core_test.cpp)
  target_link_libraries(core_test PRIVATE hyprvoice_core)
  add_test(NAME core COMMAND core_test)
  add_executable(rewrite_probe tests/rewrite_probe.cpp)
  target_link_libraries(rewrite_probe PRIVATE hyprvoice_core)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME rewrite_http_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/rewrite_http_test.py $<TARGET_FILE:rewrite_probe>)
  add_executable(test_editor tests/editor.cpp)
  target_link_libraries(test_editor PRIVATE PkgConfig::GTK)
  pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client xkbcommon)
  find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
  add_custom_command(OUTPUT virtual-keyboard.h virtual-keyboard.c
    COMMAND ${WAYLAND_SCANNER} client-header ${CMAKE_CURRENT_SOURCE_DIR}/tests/virtual-keyboard.xml ${CMAKE_CURRENT_BINARY_DIR}/virtual-keyboard.h
    COMMAND ${WAYLAND_SCANNER} private-code ${CMAKE_CURRENT_SOURCE_DIR}/tests/virtual-keyboard.xml ${CMAKE_CURRENT_BINARY_DIR}/virtual-keyboard.c
    DEPENDS tests/virtual-keyboard.xml)
  add_executable(test_keyboard tests/keyboard.cpp ${CMAKE_CURRENT_BINARY_DIR}/virtual-keyboard.c)
  target_include_directories(test_keyboard PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(test_keyboard PRIVATE PkgConfig::WAYLAND)
endif()
install(TARGETS hyprvoice RUNTIME DESTINATION bin)
