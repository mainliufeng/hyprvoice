cmake_minimum_required(VERSION 3.22)
project(hyprvoice VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4 gtk4-layer-shell-0)
pkg_check_modules(PW REQUIRED IMPORTED_TARGET libpipewire-0.3)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(ATSPI REQUIRED IMPORTED_TARGET atspi-2>=2.52)
find_library(SHERPA sherpa-onnx-c-api REQUIRED)
add_library(hyprvoice_json INTERFACE)
target_include_directories(hyprvoice_json INTERFACE third_party)
find_package(Threads REQUIRED)
add_library(hyprvoice_core src/config.cpp src/diagnostics.cpp src/context.cpp src/browser_text.cpp src/input_watch.cpp src/process.cpp src/asr.cpp src/fun_backend.cpp src/speech.cpp src/transcript_text.cpp src/audio.cpp src/desktop.cpp src/rewrite.cpp)
target_include_directories(hyprvoice_core PUBLIC src)
target_link_libraries(hyprvoice_core PUBLIC PkgConfig::GTK PkgConfig::PW PkgConfig::CURL PkgConfig::ATSPI ${SHERPA} hyprvoice_json Threads::Threads)
target_compile_options(hyprvoice_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(hyprvoice src/main.cpp src/app.cpp src/settings_panel.cpp)
target_link_libraries(hyprvoice PRIVATE hyprvoice_core)
target_compile_options(hyprvoice PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(settings_test tests/settings_test.cpp)
  target_link_libraries(settings_test PRIVATE hyprvoice_core)
  add_test(NAME settings_transaction_diagnostics COMMAND settings_test)
  add_executable(core_test tests/core_test.cpp)
  target_link_libraries(core_test PRIVATE hyprvoice_core)
  add_test(NAME core COMMAND core_test)
  add_executable(ui_text_test tests/ui_text_test.cpp)
  target_link_libraries(ui_text_test PRIVATE hyprvoice_core)
  add_test(NAME ui_text_config COMMAND ui_text_test)
  add_executable(rewrite_probe tests/rewrite_probe.cpp)
  target_link_libraries(rewrite_probe PRIVATE hyprvoice_core)
  add_executable(retry_probe tests/retry_probe.cpp)
  target_link_libraries(retry_probe PRIVATE hyprvoice_core)
  add_executable(fidelity_probe tests/fidelity_probe.cpp)
  target_link_libraries(fidelity_probe PRIVATE hyprvoice_core)
  add_executable(input_target_test tests/input_target_test.cpp)
  target_link_libraries(input_target_test PRIVATE hyprvoice_core)
  add_test(NAME input_target_snapshot COMMAND input_target_test)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME benchmark_metrics COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_benchmark_score.py)
  add_test(NAME fun_worker_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fun_worker_test.py $<TARGET_FILE:hyprvoice>)
  add_test(NAME rewrite_http_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/rewrite_http_test.py $<TARGET_FILE:rewrite_probe>)
  add_test(NAME retry_http_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/retry_http_test.py $<TARGET_FILE:retry_probe>)
  add_test(NAME fidelity_http_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fidelity_http_test.py $<TARGET_FILE:fidelity_probe> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fidelity_cases.json)
  add_executable(context_probe tests/context_probe.cpp)
  target_link_libraries(context_probe PRIVATE hyprvoice_core)
  add_executable(workflow_probe tests/workflow_probe.cpp)
  target_link_libraries(workflow_probe PRIVATE hyprvoice_core)
  add_executable(test_editor tests/editor.cpp)
  target_link_libraries(test_editor PRIVATE PkgConfig::GTK)
  pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client xkbcommon)
  find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
  add_custom_command(OUTPUT virtual-keyboard.h virtual-keyboard.c
    COMMAND ${WAYLAND_SCANNER} client-header ${CMAKE_CURRENT_SOURCE_DIR}/tests/virtual-keyboard.xml ${CMAKE_CURRENT_BINARY_DIR}/virtual-keyboard.h
    COMMAND ${WAYLAND_SCANNER} private-code ${CMAKE_CURRENT_SOURCE_DIR}/tests/virtual-keyboard.xml ${CMAKE_CURRENT_BINARY_DIR}/virtual-keyboard.c
    DEPENDS tests/virtual-keyboard.xml)
  add_executable(test_keyboard tests/keyboard.cpp ${CMAKE_CURRENT_BINARY_DIR}/virtual-keyboard.c)
  target_include_directories(test_keyboard PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(test_keyboard PRIVATE PkgConfig::WAYLAND)
endif()
install(TARGETS hyprvoice RUNTIME DESTINATION bin)
option(HYPRVOICE_WITH_FUN_ASR "Build the optional CPU Fun-ASR-Nano worker" OFF)
if(HYPRVOICE_WITH_FUN_ASR)
  include(cmake/FunRuntime.cmake)
endif()
